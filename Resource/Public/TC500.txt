statistics
    [Arguments]    ${group}    ${x}    # 统计出口组/入口组流量,${x}:入口组为1，出口组为9
    sleep    2
    ${next}=    evaluate    ${x}+1
    ${byte}=    set variable    ${0}
    ${pkt}=    set variable    ${0}
    ${group_len}=    get length    ${group}
    ${interface_statistics_pkt}=    Create list
    ${interface_statistics_byte}=    Create list
    New Connection Custom
    FOR    ${i}    IN RANGE    ${group_len}
        ${port}=    get from list    ${group}    ${i}
        telnet.write    show statistics interface ${port}
        ${information}=    Telnet.Read Until    ${userName1}(config)#
        ${statistic_list}=    statistic    ${information}
        ${single_pkt}=    get from list    ${statistic_list}    ${x}
        ${single_byte}=    get from list    ${statistic_list}    ${next}
        ${byte}=    Evaluate    ${byte}+${single_byte}
        ${pkt}=    Evaluate    ${pkt}+${single_pkt}
        append to list    ${interface_statistics_pkt}    ${single_pkt}
        append to list    ${interface_statistics_byte}    ${single_byte}
    END
    RETURN    ${interface_statistics_pkt}    ${interface_statistics_byte}    ${pkt}    ${byte}
