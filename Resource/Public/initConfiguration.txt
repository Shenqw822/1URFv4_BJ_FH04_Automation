Wait Device Reboot
    [Documentation]  冷重启后轮询等待设备上线: 直接telnet登录→show version→校验板卡与版本
    [Arguments]    ${max_wait}=660    ${interval}=60
    ${max_wait}=    Convert To Integer    ${max_wait}
    ${interval}=    Convert To Integer    ${interval}
    Sleep    120s
    ${elapsed}=    Evaluate    0
    WHILE    $elapsed < $max_wait
        TRY
            Telnet.Close All Connections
            Telnet.Open Connection    ${host_ip}    port=${cli_login_port}    timeout=10
            Telnet.Login    ${userName1}    ${userPw1}    login_prompt=user:    password_prompt=password:    login_timeout=10
            telnet.write    config
            telnet.read until    ${userName1}(config)#
            telnet.write    terminal length 0
            telnet.read until    ${userName1}(config)#
            telnet.write    show version
            ${output}    Telnet.Read Until    ${userName1}(config)#
            Log    ${output}
            # 检查software行是否有版本号填充(未填充=设备未完全启动)
            ${sw_line}=    Get Lines Matching Regexp    ${output}    software\\s+:\\s+\\S    partial_match=True
            ${sw_len}=    Get Length    ${sw_line}
            IF    ${sw_len} == 0
                # 分支1: software字段为空, 设备未完全上线, 继续等待
                Log    设备已可登录但show version版本信息为空(software字段未填充), 继续等待...
            ELSE
                # software字段有版本号: 设备已完全启动, 归一化换行符后比对
                ${cr}=    Evaluate    chr(13)
                ${lf}=    Evaluate    chr(10)
                ${crlf}=    Catenate    ${cr}${lf}
                ${output_norm}=    Replace String    ${output}    ${crlf}    ${lf}
                ${ref_norm}=    Replace String    ${device_show_version_ref}    ${crlf}    ${lf}
                ${ref_len}=    Get Length    ${ref_norm}
                IF    ${ref_len} == 0
                    Log    设备已完全上线(无初始化version基准可比对)
                    RETURN
                ELSE IF    $output_norm == $ref_norm
                    # 分支2: output与初始化version一致, 设备完全上线
                    Log    设备已完全上线, show version与初始化一致
                    RETURN
                ELSE
                    # 分支3: output非空但与初始化version不一致, 上报异常但继续执行
                    Log    设备已上线但show version与初始化不一致!
                    Log    初始化version: ${device_show_version_ref}
                    Log    当前version: ${output}
                    RETURN
                END
            END
        EXCEPT
            Log    设备尚未就绪(Telnet连接或登录失败), 继续等待...
        END
        Telnet.Close All Connections
        Sleep    ${interval}s
        ${elapsed}    Evaluate    ${elapsed} + ${interval}
        Log    等待设备上线... 已等待${elapsed}秒
    END
    Log    等待设备重启超时(${max_wait}秒)，设备未上线，继续执行后续步骤
